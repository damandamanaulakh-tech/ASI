from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter

wb = Workbook()

# Color palette
HEADER_BG = "1F2937"
HEADER_FG = "FFFFFF"
SECTION_BG = "FEF3C7"
CRITICAL_BG = "FEE2E2"
GREEN_BG = "D1FAE5"

thin = Side(border_style="thin", color="D1D5DB")
border = Border(left=thin, right=thin, top=thin, bottom=thin)

def style_header(cell):
    cell.font = Font(bold=True, color=HEADER_FG, name="Arial", size=11)
    cell.fill = PatternFill("solid", start_color=HEADER_BG)
    cell.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)
    cell.border = border

def style_cell(cell, bold=False, fill=None):
    cell.font = Font(bold=bold, name="Arial", size=10)
    cell.alignment = Alignment(wrap_text=True, vertical="top")
    cell.border = border
    if fill:
        cell.fill = PatternFill("solid", start_color=fill)

def add_sheet(name):
    ws = wb.create_sheet(name)
    return ws

def write_table(ws, headers, rows, widths=None, row_height=60):
    for i, h in enumerate(headers, 1):
        c = ws.cell(row=1, column=i, value=h)
        style_header(c)
    for r_idx, row in enumerate(rows, 2):
        for c_idx, val in enumerate(row, 1):
            c = ws.cell(row=r_idx, column=c_idx, value=val)
            style_cell(c, bold=(c_idx == 1))
    if widths:
        for i, w in enumerate(widths, 1):
            ws.column_dimensions[get_column_letter(i)].width = w
    ws.row_dimensions[1].height = 32
    for r in range(2, len(rows) + 2):
        ws.row_dimensions[r].height = row_height
    ws.freeze_panes = "A2"
