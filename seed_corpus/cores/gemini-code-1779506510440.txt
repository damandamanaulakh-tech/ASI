@dataclass
class Proposal:
    id: int
    title: str
    tier_bracket: str
    scriptural_key: str
    specification: str
    is_locked: bool = True

@dataclass
class ObserverNode:
    id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    mindset_tier: str = "60%"
    intent_channels: Dict[str, float] = field(default_factory=lambda: {
        "Kamma": 0.2, "Krode": 0.2, "Lobh": 0.2, "Moh": 0.2, "Ahankar": 0.2
    })
    internal_taqwa: float = 1.0
    sva_dharma_axis: float = 1.0
    cached_energy: float = 100.0
