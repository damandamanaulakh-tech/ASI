from docx import Document
from docx.shared import Pt, Inches, RGBColor
from docx.enum.text import WD_ALIGN_PARAGRAPH
from docx.enum.style import WD_STYLE_TYPE

def create_document(filename, title, entries):
    doc = Document()
    
    # Title
    title_para = doc.add_heading(title, 0)
    title_para.alignment = WD_ALIGN_PARAGRAPH.CENTER
    
    # Intro
    intro = doc.add_paragraph()
    intro.add_run("Full Raw Thought Extraction - Upgraded Structure\n").bold = True
    intro.add_run("Each entry contains: Definition, User said/meant, Emotional Tone, Core Belief, Key Quote, Assistant answered, How the answer changed, Intent, Hidden Layer, Best Outcome, Risk/Downside, Theory Value, Connection, Personal Stake, Actionability.")
    
    doc.add_paragraph()
    
    for i, entry in enumerate(entries, 1):
        # Raw Thought Header
        header = doc.add_heading(f"Raw Thought {i:02d} — {entry['theme']}", level=1)
        
        # Definition
        p = doc.add_paragraph()
        p.add_run("Definition: ").bold = True
        p.add_run(entry['definition'])
        
        # User said / meant
        p = doc.add_paragraph()
        p.add_run("User said / meant: ").bold = True
        p.add_run(entry['user_said'])
        
        # Emotional Tone
        p = doc.add_paragraph()
        p.add_run("Emotional Tone: ").bold = True
        p.add_run(entry['emotional_tone'])
        
        # Core Belief
        p = doc.add_paragraph()
        p.add_run("Core Belief: ").bold = True
        p.add_run(entry['core_belief'])
        
        # Key Quote
        p = doc.add_paragraph()
        p.add_run("Key Quote: ").bold = True
        p.add_run(entry['key_quote'])
        
        # Assistant answered
        p = doc.add_paragraph()
        p.add_run("Assistant answered: ").bold = True
        p.add_run(entry['assistant_answered'])
        
        # How the answer changed
        p = doc.add_paragraph()
        p.add_run("How the answer changed: ").bold = True
        p.add_run(entry['how_changed'])
        
        # Intent
        p = doc.add_paragraph()
        p.add_run("Intent: ").bold = True
        p.add_run(entry['intent'])
        
        # Hidden Layer
        p = doc.add_paragraph()
        p.add_run("Hidden Layer: ").bold = True
        p.add_run(entry['hidden_layer'])
        
        # Best Outcome
        p = doc.add_paragraph()
        p.add_run("Best Outcome: ").bold = True
        p.add_run(entry['best_outcome'])
        
        # Risk / Downside
        p = doc.add_paragraph()
        p.add_run("Risk / Downside: ").bold = True
        p.add_run(entry['risk_downside'])
        
        # Theory Value
        p = doc.add_paragraph()
        p.add_run("Theory Value: ").bold = True
        p.add_run(entry['theory_value'])
        
        # Connection
        p = doc.add_paragraph()
        p.add_run("Connection: ").bold = True
        p.add_run(entry['connection'])
        
        # Personal Stake
        p = doc.add_paragraph()
        p.add_run("Personal Stake: ").bold = True
        p.add_run(entry['personal_stake'])
        
        # Actionability
        p = doc.add_paragraph()
        p.add_run("Actionability: ").bold = True
        p.add_run(entry['actionability'])
        
        doc.add_paragraph("─" * 60)
    
    doc.save(filename)
    print(f"Created: {filename}")
